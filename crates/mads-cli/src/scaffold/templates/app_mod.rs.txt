// src/app/mod.rs
mod controller;
mod routes;
mod service;

use mads::prelude::*;

#[furnace]
pub struct AppModule;

impl Furnace for AppModule {
    fn register(self) -> FurnaceRegistration<Self> {
        self.provide::<service::AppService>()
            .controller::<controller::AppController>()
    }
}
