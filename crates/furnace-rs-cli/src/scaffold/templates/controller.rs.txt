// src/app/controller.rs
use furnace::prelude::*;

use super::service::AppService;

#[controller]
pub struct AppController {
    service: AppService,
}

impl Sealable for AppController {
    fn seals() -> SealRegistration<Self> {
        SealRegistration::new()
    }
}

#[controller(route = "/")]
impl AppController {
    #[get]
    async fn hello(&self) -> &'static str {
        self.service.hello()
    }
}
